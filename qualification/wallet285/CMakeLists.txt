cmake_minimum_required(VERSION 3.24)
project(Wallet285 LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
foreach(t baseline candidate)
 add_executable(${t} ${root}/tests/managed_core_profile_test.cpp ${root}/src/wallet/core_profile.cpp)
 target_compile_features(${t} PRIVATE cxx_std_20)
 target_link_libraries(${t} PRIVATE Qt6::Core)
endforeach()
target_include_directories(baseline PRIVATE base ${root}/src)
target_include_directories(candidate PRIVATE ${root}/src)
enable_testing()
add_test(NAME old-release-reproduces-broken-path COMMAND baseline --baseline)
add_test(NAME new-release-reopens-with-same-wallet COMMAND candidate)
