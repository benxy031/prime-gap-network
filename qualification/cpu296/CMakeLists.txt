cmake_minimum_required(VERSION 3.24)
project(Cpu296 LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
foreach(t baseline candidate)
 add_executable(${t} ${root}/tests/cpu_normal_resources_test.cpp ${root}/src/mining/cpu_controller_client.cpp)
 target_compile_features(${t} PRIVATE cxx_std_20)
 target_include_directories(${t} PRIVATE ${root}/src)
 target_link_libraries(${t} PRIVATE Qt6::Core)
endforeach()
target_include_directories(baseline BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/base)
enable_testing()
add_test(NAME old-misrouting-reproduced COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/expect_old.py $<TARGET_FILE:baseline> ${Python3_EXECUTABLE} ${root}/tests/fixtures/cpu_normal_endpoint.py ${CMAKE_CURRENT_BINARY_DIR}/baseline-data)
add_test(NAME normal-miner-preferred-with-fallback COMMAND candidate ${Python3_EXECUTABLE} ${root}/tests/fixtures/cpu_normal_endpoint.py ${CMAKE_CURRENT_BINARY_DIR}/candidate-data)
set_tests_properties(old-misrouting-reproduced normal-miner-preferred-with-fallback PROPERTIES TIMEOUT 35)
