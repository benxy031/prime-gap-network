cmake_minimum_required(VERSION 3.24)
project(Statistics309 LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(core "${CMAKE_CURRENT_SOURCE_DIR}/../../wallet/riecoin-core-upstream-v2606/src")
add_executable(utc_period_probe utc_period_probe.cpp)
add_executable(rosettes_test ${core}/pgn/leaderboard_rosettes_test.cpp ${core}/pgn/merit_arithmetic.cpp ${core}/univalue/lib/univalue.cpp ${core}/univalue/lib/univalue_get.cpp ${core}/univalue/lib/univalue_read.cpp ${core}/univalue/lib/univalue_write.cpp)
foreach(t utc_period_probe rosettes_test)
 target_compile_features(${t} PRIVATE cxx_std_20)
 target_include_directories(${t} PRIVATE ${core} ${core}/univalue/include)
endforeach()
enable_testing()
add_test(NAME independent-gregorian-utc-oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/utc_oracle.py $<TARGET_FILE:utc_period_probe>)
add_test(NAME exact-rosettes-and-largest-gap COMMAND rosettes_test ${core}/pgn/leaderboard_rosettes_oracle.txt ${core}/pgn/leaderboard_rosettes_batch_oracle.txt)
set_tests_properties(independent-gregorian-utc-oracle exact-rosettes-and-largest-gap PROPERTIES TIMEOUT 45)
option(PGN_STATS_QT "Qualify the actual Qt client and offscreen view" OFF)
if(PGN_STATS_QT)
 set(CMAKE_AUTOMOC ON)
 set(CMAKE_AUTORCC ON)
 find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
 set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
 add_library(statistics_widgets OBJECT ${root}/src/ui/social_view.cpp ${root}/src/ui/social_access_dialog.cpp ${root}/src/ui/social_forum_thread.cpp ${root}/src/social/social_core_service.cpp ${root}/assets/pgn.qrc)
 target_include_directories(statistics_widgets PRIVATE ${root}/src ${root}/tests)
 target_compile_features(statistics_widgets PRIVATE cxx_std_20)
 target_link_libraries(statistics_widgets PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
 add_executable(utc_qt_probe utc_qt_probe.cpp)
 add_executable(qt_statistics_test qt_statistics_test.cpp $<TARGET_OBJECTS:statistics_widgets>)
 add_executable(social_existing_test ${root}/tests/social_view_test.cpp $<TARGET_OBJECTS:statistics_widgets>)
 foreach(t utc_qt_probe qt_statistics_test social_existing_test)
  target_include_directories(${t} PRIVATE ${root}/src ${root}/tests ${core})
  target_compile_features(${t} PRIVATE cxx_std_20)
  target_link_libraries(${t} PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
 endforeach()
 add_test(NAME independent-qt-native-utc-oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/utc_oracle.py $<TARGET_FILE:utc_qt_probe>)
 add_test(NAME actual-qt-statistics-controls COMMAND qt_statistics_test ${CMAKE_CURRENT_BINARY_DIR}/statistics309.png)
 add_test(NAME existing-social-behavior COMMAND social_existing_test)
 set_tests_properties(independent-qt-native-utc-oracle actual-qt-statistics-controls existing-social-behavior PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
