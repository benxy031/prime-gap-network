cmake_minimum_required(VERSION 3.24)
project(Access308 LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
foreach(variant original fixed)
 if(variant STREQUAL "original")
  set(impl "${CMAKE_CURRENT_SOURCE_DIR}/baseline")
 else()
  set(impl "${root}/src/ui")
 endif()
 add_executable(access_${variant} parent_refresh_test.cpp ${impl}/social_view.cpp ${impl}/social_access_dialog.cpp ${root}/src/ui/social_forum_thread.cpp ${root}/src/social/social_core_service.cpp ${root}/assets/pgn.qrc)
 target_compile_definitions(access_${variant} PRIVATE ORIGINAL_BEHAVIOR=$<STREQUAL:${variant},original>)
 target_include_directories(access_${variant} PRIVATE ${root}/src ${root}/tests)
 target_compile_features(access_${variant} PRIVATE cxx_std_20)
 target_link_libraries(access_${variant} PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
endforeach()
enable_testing()
add_test(NAME original-reproduces-invalidated-editor COMMAND access_original)
add_test(NAME fixed-preserves-editor-and-revocation COMMAND access_fixed)
set_tests_properties(original-reproduces-invalidated-editor fixed-preserves-editor-and-revocation PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
