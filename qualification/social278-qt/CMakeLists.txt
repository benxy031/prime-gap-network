cmake_minimum_required(VERSION 3.24)
project(Social278 LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(qt_roles_test qt_roles_test.cpp ${root}/src/ui/social_access_dialog.cpp)
add_executable(social_view_test ${root}/tests/social_view_test.cpp ${root}/src/ui/social_view.cpp ${root}/src/ui/social_access_dialog.cpp ${root}/src/ui/social_forum_thread.cpp ${root}/src/social/social_core_service.cpp ${root}/assets/pgn.qrc)
foreach(t qt_roles_test social_view_test)
 target_include_directories(${t} PRIVATE ${root}/src ${root}/tests)
 target_compile_features(${t} PRIVATE cxx_std_20)
 target_link_libraries(${t} PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
endforeach()
enable_testing()
add_test(NAME social-view COMMAND social_view_test)
set_tests_properties(social-view PROPERTIES TIMEOUT 60)
