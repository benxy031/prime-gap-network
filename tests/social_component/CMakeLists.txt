cmake_minimum_required(VERSION 3.22)
project(PGNSocialComponent LANGUAGES CXX)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
enable_testing()
add_library(social-component STATIC ../../src/ui/social_view.cpp ../../src/ui/social_access_dialog.cpp ../../src/ui/social_forum_thread.cpp ../../src/social/social_core_service.cpp)
target_include_directories(social-component PUBLIC ../../src)
target_compile_features(social-component PUBLIC cxx_std_20)
target_link_libraries(social-component PUBLIC Qt6::Widgets)
add_executable(pgn-social-view-test ../social_view_test.cpp ../../assets/pgn.qrc)
target_link_libraries(pgn-social-view-test PRIVATE social-component)
add_executable(pgn-social-native-witness ../social_native_witness.cpp ../../assets/pgn.qrc)
target_link_libraries(pgn-social-native-witness PRIVATE social-component Qt6::Network)
add_test(NAME social-view COMMAND pgn-social-view-test)
set_tests_properties(social-view PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
